cmake_minimum_required(VERSION 3.24)
project(DualForge VERSION 0.1.0 LANGUAGES C CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
option(DUALFORGE_BUILD_TESTS "Build automated tests" ON)
option(DUALFORGE_BUILD_EXAMPLES "Build CLI and standalone examples" ON)
option(DUALFORGE_BUILD_HARDWARE_CHECK "Build manual real-device output check (not run by CTest)" OFF)
option(DUALFORGE_SANITIZERS "Enable address/undefined sanitizers" OFF)
set(DUALFORGE_SDL_SOURCE_DIR "" CACHE PATH "Offline SDL 3.4.18 source directory")
set(SDL_SHARED ON CACHE BOOL "" FORCE)
set(SDL_STATIC OFF CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_INSTALL ON CACHE BOOL "" FORCE)
if(DUALFORGE_SDL_SOURCE_DIR)
  file(STRINGS "${DUALFORGE_SDL_SOURCE_DIR}/include/SDL3/SDL_version.h" _sdl_version_lines REGEX "^#define SDL_(MAJOR|MINOR|MICRO)_VERSION")
  foreach(_part MAJOR MINOR MICRO)
    string(REGEX MATCH "SDL_${_part}_VERSION +([0-9]+)" _match "${_sdl_version_lines}")
    set(_sdl_${_part} "${CMAKE_MATCH_1}")
  endforeach()
  if(NOT "${_sdl_MAJOR}.${_sdl_MINOR}.${_sdl_MICRO}" STREQUAL "3.4.18")
    message(FATAL_ERROR "Offline SDL source must be version 3.4.18")
  endif()
  add_subdirectory("${DUALFORGE_SDL_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/sdl")
else()
  include(FetchContent)
  FetchContent_Declare(SDL3
    URL https://www.libsdl.org/release/SDL3-3.4.18.tar.gz
    URL_HASH SHA256=9c75cf16330322c217dedd2e0609f1124f1b54b8633e763467b4684d0f4334a3
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(SDL3)
endif()
if(NOT SDL3_VERSION STREQUAL "3.4.18" AND DEFINED SDL3_VERSION)
  message(FATAL_ERROR "DualForge requires the verified SDL 3.4.18 source")
endif()
add_library(dualforge STATIC src/controller.cpp src/sdl_backend.cpp src/trigger_effects.cpp src/protocol.cpp)
add_library(DualForge::DualForge ALIAS dualforge)
set_target_properties(dualforge PROPERTIES EXPORT_NAME DualForge POSITION_INDEPENDENT_CODE ON)
target_compile_features(dualforge PUBLIC cxx_std_17)
target_include_directories(dualforge PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}> PRIVATE src)
target_link_libraries(dualforge PUBLIC SDL3::SDL3)
if(MSVC)
  target_compile_options(dualforge PRIVATE /W4 /permissive-)
else()
  target_compile_options(dualforge PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(DUALFORGE_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(dualforge PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(dualforge PUBLIC -fsanitize=address,undefined)
endif()
if(DUALFORGE_BUILD_EXAMPLES)
  foreach(example main test_led test_trigger_effects bow_simulation touchpad)
    if(example STREQUAL "main")
      set(_example_target dualforge_cli)
    else()
      set(_example_target dualforge_${example})
    endif()
    add_executable(${_example_target} examples/${example}.cpp)
    target_link_libraries(${_example_target} PRIVATE DualForge::DualForge)
    if(MSVC)
      target_compile_options(${_example_target} PRIVATE /utf-8 /W4)
    else()
      target_compile_options(${_example_target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    if(WIN32)
      add_custom_command(TARGET ${_example_target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:SDL3::SDL3> $<TARGET_FILE_DIR:${_example_target}>)
    endif()
  endforeach()
  target_sources(dualforge_touchpad PRIVATE examples/mouse_output.cpp)
  if(WIN32)
    target_link_libraries(dualforge_touchpad PRIVATE user32)
  elseif(APPLE)
    find_library(DUALFORGE_APPLICATION_SERVICES ApplicationServices REQUIRED)
    target_link_libraries(dualforge_touchpad PRIVATE ${DUALFORGE_APPLICATION_SERVICES})
  endif()
endif()
if(DUALFORGE_BUILD_HARDWARE_CHECK)
  add_executable(dualforge_hardware_check tests/hardware_smoke.cpp)
  target_link_libraries(dualforge_hardware_check PRIVATE DualForge::DualForge)
  if(WIN32)
    add_custom_command(TARGET dualforge_hardware_check POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:SDL3::SDL3> $<TARGET_FILE_DIR:dualforge_hardware_check>)
  endif()
endif()
if(DUALFORGE_BUILD_TESTS)
  enable_testing()
  foreach(name protocol lifecycle sdl gestures)
    add_executable(test_${name} tests/test_${name}.cpp)
    target_include_directories(test_${name} PRIVATE src tests examples)
    target_link_libraries(test_${name} PRIVATE DualForge::DualForge)
    if(MSVC)
      target_compile_options(test_${name} PRIVATE /utf-8)
    endif()
    add_test(NAME ${name} COMMAND test_${name})
    set_tests_properties(${name} PROPERTIES ENVIRONMENT "SDL_VIDEO_DRIVER=dummy;SDL_AUDIO_DRIVER=dummy")
    if(WIN32)
      add_custom_command(TARGET test_${name} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:SDL3::SDL3> $<TARGET_FILE_DIR:test_${name}>)
    endif()
  endforeach()
  if(DUALFORGE_BUILD_EXAMPLES)
    foreach(example test_led test_trigger_effects bow_simulation touchpad)
      add_test(NAME help_${example} COMMAND dualforge_${example} --help)
    endforeach()
  endif()
endif()
install(TARGETS dualforge EXPORT DualForgeTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES LICENSE THIRD_PARTY_NOTICES.md DESTINATION ${CMAKE_INSTALL_DATADIR}/dualforge)
install(EXPORT DualForgeTargets NAMESPACE DualForge:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/DualForge)
configure_package_config_file(cmake/DualForgeConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/DualForgeConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/DualForge)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/DualForgeConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/DualForgeConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/DualForgeConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/DualForge)
